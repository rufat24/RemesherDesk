set(RDESK_PLATFORM_SOURCES src/process.cpp src/utf.cpp src/paths.cpp)
if(WIN32)
  list(APPEND RDESK_PLATFORM_SOURCES src/process_win32.cpp)
else()
  list(APPEND RDESK_PLATFORM_SOURCES src/process_posix.cpp)
endif()
find_package(Threads REQUIRED)
add_library(rdesk_platform ${RDESK_PLATFORM_SOURCES})
add_library(rdesk::platform ALIAS rdesk_platform)
target_include_directories(rdesk_platform PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_link_libraries(rdesk_platform PUBLIC rdesk::core Threads::Threads)
rdesk_target_defaults(rdesk_platform)
